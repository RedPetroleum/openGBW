# The noise measurement of the Calibrate submenu: the raw readings of the resting scale drawn as the
# distribution of their distance from their own average. "noise" gives the simulated scale the scatter
# a real load cell has, otherwise every reading would be the same number
[noise measurement]
noise 0.065
click; turn 1; click; turn 5; click # menu -> Calibrate -> Noise, the click starts the bar
wait 2000; shot noise-early # the first readings, the scale of the drawing is still finding itself
wait 8000; shot noise-half  # the bell fills up while the bar above it runs out
wait 11000; shot noise      # the bar is gone, the sigma stands in its place and collecting goes on
wait 15000; shot noise-long # left standing longer, the bell only gets smoother
click; shot noise-save      # a click or a turn asks what to do with the sigma
turn 1; shot noise-discard  # turning switches between the two
turn 1; click # back on Save and taken, the sigma stands in the Info Menu from here on
