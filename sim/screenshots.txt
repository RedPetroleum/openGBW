# Screenshots of all screens, run with "make screenshots" in sim/
# Every [section] starts from a freshly started firmware. Commands: see README.md

# The initializing screen fills over BOOT_READY_MS and then stands still, so every one of these shots
# needs its own section: two "boot" commands in one would go on where the first one stopped
[boot]
boot 700; shot boot

[boot, turned]
boot 1500; shot boot-turned # the cup turns while the border grows

[boot, almost ready]
boot 2900; shot boot-ready # shortly before the border is closed

[boot, as a bar]
set bootScreenStyle 0; boot 1500; shot boot-bar # the other styles of the Style menu

[boot, as a cross]
set bootScreenStyle 2; boot 750; shot boot-cross # turns a quarter clockwise until the scale is ready

[main screen]
shot main

[grinding]
set cupWeightEmpty 396.1; weight 405.3; set scaleStatus 1; set grindTime 4.2
wait 600; set grindTime 4.2 # the progress bar follows the reading smoothly, give it time to catch up
draw; shot grinding
set scaleStatus 6; weight 412.5; set grindTime 7.9
wait 600; set grindTime 7.9; draw; shot verifying # grinder off, waiting for a steady reading
set scaleStatus 3; set grindFailReason Cup removed

# The finished screen averages the raw readings from the moment the dose was confirmed, so each of
# these sections feeds a second of readings and then enters the screen with "verifiedAgo"
